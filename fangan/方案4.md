```typescript
// 新文件: src/api/ai/component-knowledge-builder.ts
export async function buildComponentKnowledgeBase() {
  // 动态扫描所有组件
  const components = import.meta.glob('/src/views/web-list/**/*.vue', { 
    query: '?raw', import: 'default' 
  })

  const knowledge: ComponentKnowledge[] = []
  
  for (const [path, loader] of Object.entries(components)) {
    const source = await loader() as string
    const readme = await loadReadme(path)  // 读取同目录README
    
    knowledge.push({
      path,
      name: extractComponentName(path),
      category: extractCategory(path),     // card-image/card-text/card-3d...
      summary: extractSummary(readme, 80),  // README 前80字
      features: extractFeatures(readme),    // 从README提取核心特性
      props: extractProps(source),          // 从源码提取 Props/参数
      gsapUsage: extractGSAP(source),        // 提取GSAP/ScrollTrigger用法
      visualStyle: classifyStyle(readme),    // 分类为: neon/cyber/luxury/minimal...
    })
  }
  
  return knowledge  // ~50K tokens，完全动态
}
```

然后注入给 DeepSeek：
```typescript
async generateCodeWithContext(userRequest: string) {
  const kb = await buildComponentKnowledgeBase()
  
  const response = await this.chat({
    messages: [
      { 
        role: 'system', 
        content: `
你是GSAP动画网站代码生成专家。以下是全部${kb.length}个可用组件：

${kb.map(c => 
  `### ${c.name} (${c.category})
  - 风格: ${c.visualStyle}
  - 特性: ${c.features}
  - 适用: ${c.summary}
  - Props: ${JSON.stringify(c.props)}
  `
).join('\n')}

你需要直接生成完整的React+Tailwind组件代码。`
      },
      { role: 'user', content: userRequest }
    ],
    maxTokens: 16000,  // 从4000提升到16000
    stream: true
  })
}
```
